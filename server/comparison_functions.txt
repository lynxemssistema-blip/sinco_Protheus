// Add after fetchHistory function:
const fetchComparisonData = async () => {
    setLoading(true);
    try {
        const params = {
            ...filters
        };

        const response = await axios.get('/api/production/tracking-comparison', { params });
        setComparisonSummary(response.data.resumo);
        setComparisonData(response.data.operacoes);
    } catch (error) {
        console.error('Error fetching comparison data:', error);
        alert('Erro ao carregar dados de comparação');
    } finally {
        setLoading(false);
    }
};

// Add useEffect for comparison tab:
useEffect(() => {
    if (activeTab === 'comparison') {
        fetchComparisonData();
    }
}, [activeTab]);

// Status badge component:
const StatusBadge = ({ status }) => {
    const statusConfig = {
        complete: { color: 'bg-green-100 text-green-800', icon: '✓', text: 'Completo' },
        partial: { color: 'bg-yellow-100 text-yellow-800', icon: '⚠️', text: 'Parcial' },
        low: { color: 'bg-orange-100 text-orange-800', icon: '⚠️', text: 'Baixo' },
        pending: { color: 'bg-gray-100 text-gray-800', icon: '⏳', text: 'Pendente' },
        extra: { color: 'bg-blue-100 text-blue-800', icon: '➕', text: 'Extra' }
    };
    
    const config = statusConfig[status] || statusConfig.pending;
    
    return (
        <span className={`px-2 py-1 rounded-full text-xs font-semib old ${config.color}`}>
            {config.icon} {config.text}
        </span>
    );
};
